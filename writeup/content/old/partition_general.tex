\section{Recovering $k$-partitions from well-behaved model matrices}\label{sec: partition_general}

\begin{lemma}[Non-uniformity of eigenvectors]\label{lem: eigenvectorsdistinguish}
    Let $D$ be a $k\times k$ diagonal matrix with nonnegative entries, where we denote by $D(x) = D_{xx}$.
    Suppose $D(x) \leq \e$ for all $x \in [k]$, for some $\gamma > 0$ small.
    % are positive on the diagonal such that $D(x) \leq \frac{1}{\sqrt{\gamma}}$ for all $x \in [n]$, and 
    Let $M$ be a symmetric $k\times k$ matrix with $\Norm{M}=1$ such that for any two distinct columns $M_x$ and $M_y$ it holds that $\Norm{D(x)M_x-D(y)M_y}^2 \geq \alpha^2$.
    Let $v_1, \ldots, v_k$ be orthonormal eigenvectors of $M$ and let $S:=\Set{v_r \; \mid\; M v_r = \tau v_r, |\tau| > \gamma}$.
    Then,
    $$\min_{x\neq y\in[k]}\max_{v\in S} \Abs{D(x) v_x-D(y) v_y}\geq \sqrt{\frac{\alpha^2-2\e^2\gamma^2}{k\Paren{1-\gamma^2}}} \,.$$
\end{lemma}

\begin{proof}
    We write $M = \sum_{r=1}^k \lambda_r v_r v_r^{\top}$ and get that, for any $x \neq y \in [k]$,
    \begin{equation*}
        D(x) M_x - D(y) M_y=\sum_{r=1}^k \lambda_r (D(x) (v_r)_x - D(y) (v_r)_y)v_r \,.
    \end{equation*}
    Using that $\Norm{M}=1$ and $\Abs{\lambda_r}\leq \gamma$, we get
    \begin{align}
    \label{eq:rounding_split_1}
        \begin{split}
            \Norm{D(x) M_x- D(y) M_y}^2
            &=\sum_{r=1}^k \lambda_r^2 (D(x) (v_r)_x - D(y)(v_r)_y)^2 \\
            &=\sum_{r\in S} \lambda_r^2 (D(x) (v_r)_x - D(y)(v_r)_y)^2+\sum_{r\notin S} \lambda_r^2 (D(x) (v_r)_x - D(y)(v_r)_y)^2 \\
            &\leq \sum_{r\in S} (D(x) (v_r)_x- D(y) (v_r)_y)^2+\gamma^2\sum_{r\notin S} (D(x) (v_r)_x- D(y) (v_r)_y)^2 \,.
        \end{split}
    \end{align}
    Because the eigenvectors are orthonormal, the rows of the matrix $V \in \R^{k \times k}$ whose columns are $\set{v_{r}}_{r \in [k]}$ are also orthonormal: that is, $\sum_{r=1}^{k} (v_r)_x (v_r)_y = 0$ for $x \neq y$ and $\sum_{r=1}^{k} (v_r)_x^2 = 1$ for $x \in [k]$.
    Therefore, we obtain
    \begin{align*}
        \sum_{r \notin S} (D(x) (v_r)_x- D(y) (v_r)_y)^2
        & = \sum_{r=1}^k (D(x) (v_r)_x- D(y) (v_r)_y)^2 - \sum_{r \in S} (D(x) (v_r)_x- D(y) (v_r)_y)^2 \\
        & = D(x)^2 + D(y)^2 - \sum_{r \in S} (D(x) (v_r)_x- D(y) (v_r)_y)^2 \\
        % & \leq D(x)^2 + D(y)^2 - \sum_{r \in S} (D(x) (v_r)_x- D(y) (v_r)_y)^2 \\
        & \leq 2\e^2 - \sum_{r \in S} (D(x) (v_r)_x- D(y) (v_r)_y)^2 \,.
    \end{align*}
    Plugging it into \cref{eq:rounding_split_1}, we get
    \begin{equation*}
        \Norm{D(x) M_x- D(y) M_y}^2
        \leq \Paren{1-\gamma^2} \sum_{r \in S} (D(x) (v_r)_x- D(y) (v_r)_y)^2 + 2 \e^2 \gamma^2 \,.
    \end{equation*}
    Since $\Norm{D(x)M_x-D(y)M_y}^2\geq \alpha^2$, it follows that
    \begin{equation*}
        \sum_{r \in S} (D(x) (v_r)_x- D(y) (v_r)_y)^2 \geq \frac{\alpha^2 - 2\e^2\gamma^2}{1-\gamma^2} \,.
    \end{equation*}
    By an averaging argument, there must exist some $r \in S$ such that
    \begin{equation*}
        (D(x) (v_r)_x- D(y) (v_r)_y)^2 \geq \frac{\alpha^2 - 2\e^2\gamma^2}{k(1-\gamma^2)} \,.
    \end{equation*}
\end{proof}

\begin{algorithm}[H]
    \caption{Spectral partitioning algorithm (Based on Algorithm 10 of \cite{MR3536556-David16})}
    \label{alg: spectralclustering}
    \DontPrintSemicolon
    For every possible $k$-tuple of vertices $x_{[k]}\in V$ try to following partition:\;
    Let $S_1 = \{x_1\}, \ldots, S_k = \{ x_k\}$ and for all other vertices $y \in V$ add $y$ into the partition $S_{i}$ that minimizes
    \[
    \sum_{j=1}^l \Paren{\hat{u}_j(y)-\hat{u}_j(x_i)}^2\,.
    \]
\end{algorithm}


\begin{lemma}[Partitioning based on distinguishing eigenvectors, Based on Lemma E.7 of \cite{MR3536556-David16}]\label{lem:clusteringalmosteigenvectors}
    Let $G$ be a graph on $n$ vertices $V$. Let $\set{v_{i}}_{i \in [l]}$ be $k$-dimensional vectors such that $\min_{a\neq b\in[k]}\max_{i\in [l]} \Abs{v_i(a)-v_i(b)}\geq \frac{\zeta}{\sqrt{n}}$. Let $\set{\hat{u}_{i}}_{i \in [l]}$ be $n$-dimensional vectors such that $\sum_{j=1}^l \Norm{\hat{u}_j-Zv_j}^2\leq \eta$.
    Then, \cref{alg: spectralclustering} runs in time $\poly(n)$ and returns a list of $\poly(n)$ partitions of $V$ such that one of them has at most $\frac{12k\eta}{\zeta^2}n$ vertices wrongly partitioned.
\end{lemma}

\begin{proof}
    The running time is $n^{O(k)}$.
    
    Let $u_j:=Zv_j$ and note that if $Z_{ia}=1$ then $u_j(i)=v_j(a)$. Let us define a set
    
    $$\mathrm{Good}:=\Set{x\in V\;\mid\; \Paren{\hat{u}_j(x)-u_j(x)}^2\leq \frac{\zeta^2}{12kn}}\,.$$
    
    Assume that $x_a$ is a vertex chosen at step 1 of \cref{alg: spectralclustering} such that $x_a\in\mathrm{Good}$ and $Z_{x_a a}=1$, and $y\in\mathrm{Good}$ is a vertex with $Z_{ya}=1$ to be partitioned. Then, note that $u_j(x_a)=v_j(a)=u_j(y)$, $\Paren{\hat{u}_j(x_a)-u_j(x_a)}^2\leq \frac{\zeta^2}{12kn}$ and $\Paren{\hat{u}_j(y)-u_j(y)}^2\leq \frac{\zeta^2}{12kn}$. Hence, for any $j\in[l]$ writing $\Paren{\hat{u}_j(y)-\hat{u}_j(x_a)}^2$ as
    
    $$\Bigg(\Paren{\hat{u}_j(x_a)-u_j(x_a)}+\Paren{u_j(x_a)-u_j(y)}+\Paren{u_j(y)-\hat{u}_j(y)}\Bigg)^2$$
    
    and using $(p+q)^2\leq 2(p^2+q^2)$ we get

    $$\sum_{j=1}^l \Paren{\hat{u}_j(y)-\hat{u}_j(x_a)}^2\leq 2l \frac{\zeta^2}{12kn}=\frac{\zeta^2}{6n}\,.$$

    On the other hand, let $x_b$ be another vertex chosen at step 1 of \cref{alg: spectralclustering} such that $x_b\in\mathrm{Good}$ and $Z_{x_b b}=1$ for some $b\neq a$. Then, by assumption there exists $i\in[l]$ with $u_i(x_b)-u_i(y)=\Abs{v_i(b)-v_i(a)}\geq \frac{\zeta}{\sqrt{n}}$, so similar to before writing $\Paren{\hat{u}_i(y)-\hat{u}_i(x_b)}^2$ as
    
    $$\Bigg(\Paren{\hat{u}_i(x_b)-u_i(x_b)}+u_i(x_b)-u_i(y)+\Paren{u_i(y)-\hat{u}_i(y)}\Bigg)^2$$

    and using $\frac{\zeta}{\sqrt{n}}\geq 2\sqrt{\frac{\zeta^2}{12kn}}$ we get

    $$\sum_{j=1}^l \Paren{\hat{u}_j(y)-\hat{u}_j(x_b)}^2\geq \Paren{\hat{u}_i(y)-\hat{u}_i(x_b)}^2 \geq \Paren{\frac{\zeta}{\sqrt{n}}-2\sqrt{\frac{\zeta^2}{12kn}}}^2>\frac{\zeta^2}{6n}\, ,$$


    showing that when the $x_{[k]}$ vertices are all in $\mathrm{Good}$ and align with $Z$, all vertices in $\mathrm{Good}$ get correctly partitioned. In this case, at most $\frac{\eta}{\frac{\zeta^2}{12kn}}=\frac{12k\eta}{\zeta^2}n$ vertices get wrongly partitioned, finishing the proof.
\end{proof}

\begin{theorem}[Restatement of theorem for partition recovery]
    Fix $k \geq 2, t \geq 1$ integers and $c > 0$.
    For all $\delta > 0$, there exist $\e, \lambda > 0$ such that the following holds.
    Let $G$ be a regular graph and let $\chi: [n] \to [k]$ be a $k$-coloring of its vertices.
    Let $M = M(G, \chi)$ be its model.
    %, and suppose its minimum non-zero eigenvalue is larger than $c$ in absolute value. \rares{Too strong}
    % Let $W = Z M B^{-1} Z^\top \in \R^{n \times n}$.
    Suppose the following hold:
    \begin{itemize}
        \item Any two rows of $M$ satisfy $\Norm{\Paren{M^{(a)} - M^{(b)}}} \geq c$.
        \item For $W = Z M B^{-1} Z^\top$, it holds for all $a \in [k]$ that  $\Norm{\tilde{A} Z_a - W  Z_a}^2 \leq \e n$.
        \item $\rank_{\geq \lambda}(\tilde{A}) \leq t$.
        \item $|\chi^{-1}(a)| \geq c n$ for all $a \in [k]$.
    \end{itemize}
    Then there exists an algorithm that runs in polynomial time and outputs some $\hat{\chi}: [n] \to [k]$ such that, up to a permutation of $[k]$, we have for a uniform random vertex $\bm{x} \sim [n]$ that $\mathbb{E}_{\bm{x} \sim \operatorname{Unif}([n])} \1[\hat\chi(x) \neq \chi(x)] \leq \delta$.
\end{theorem}

\begin{proof}[Proof of \cref{thm:part-recovery}]
    We note that $W$ has an eigenvector $\1$ with eigenvalue $1$.
    Define now the set $S:=\Set{u \in \R^n\; \mid\; W u = \tau u, \Abs{\tau}>\lambda + \eta}$ of eigenvectors of $W$ larger than $\lambda + \eta$ in absolute value.
    By \cref{thm:partition_recovery_theorem2}, for each unit eigenvector $u$ in this set we recover some $\hat{u}$ such that $\norm{\hat{u}-u}^2 \leq O(\frac{\e}{\eta^2 c})$, in time $n^{O(1)} \cdot \Paren{\frac{\e}{\eta^2 c}}^{O(t/\lambda^2)}$.

    Observe that each eigenvector $u$ of $W$ is by definition in the column span of $Z$, so we can write $u = Z v$ for some $v \in \R^{k}$.
    Let $Wu = \tau u$.
    Then $M v = B^{-1} Z^\top W Z v = \tau B^{-1} Z^\top Z v = \tau v$, so $v$ is an eigenvector of $M$ with eigenvalue $\tau$.
    Moreover, we observe that if $Mv = \tau v$, then we also have that $B^{1/2}v$ is an eigenvector of $\tilde{M} = B^{1/2} M B^{-1/2}$ of value $\tau$.
    Here $\tilde{M}$ is symmetric and has top eigenvalue $1$.

    Then, we apply \cref{lem: eigenvectorsdistinguish} with the matrix $\tilde{M}$ and with the vectors $B^{1/2} v$ normalized to be unit, where $v$ corresponds to eigenvectors $u \in S$ via the transformation $u = Z v$.
    We apply the lemma with $D = \sqrt{n} B^{-1/2}$.
    Recall that we assume any two rows of $M$ satisfy $\Norm{M^{(a)} - M^{(b)}} \geq c$, so also any two rows of $M B^{-1/2} = B^{-1/2} \tilde{M}$ satisfy $\Norm{\sqrt{n}B^{-1/2} \Paren{M^{(a)} - M^{(b)}}} \geq c$.
    Taking the transpose and using that $\tilde{M}$ is symmetric, any two columns of $\tilde{M} B^{-1/2}$ satisfy $\Norm{\sqrt{n}B^{-1/2}(a) \tilde{M}_a - \sqrt{n}B^{-1/2}(b) \tilde{M}_b} \geq c$.
    Applying \cref{lem: eigenvectorsdistinguish}, we get 
    \[\min_{a\neq b\in[k]}\max_{\stackrel{Zv \in S}{\Norm{B^{1/2}v}=1}} \sqrt{n} \Abs{v_a- v_b} \geq \Omega\Paren{\sqrt{c^2 - 2(\lambda+\eta)^2/c}} \geq \Omega(c)\,.\]

    The above shows that, for all $a \neq b \in [k]$, there exists some $v$ with $Zv \in S$ and $\Norm{B^{1/2}v}=1$ such that $\sqrt{n}\Abs{v_a - v_b} \geq \Omega(c)$, so $\Abs{v_a - v_b} \geq \Omega(c/\sqrt{n})$.
    We note that $\Norm{B^{1/2}v} = 1$ implies that $1/\sqrt{n} \leq \Norm{v} \leq 1/\sqrt{c n}$.
    For $u = Z v$, we also have $\sqrt{c n} \norm{v} \leq \norm{u} \leq \sqrt{n} \norm{v}$, so $\sqrt{c} \leq \norm{u} \leq 1/\sqrt{c}$.
    Then there exists some $\hat{u}$ in the net with $\norm{\hat{u} - u}^2 \leq O(\frac{\e}{\eta^2 c})$.
    Leta $\eta > 0$ be a small enough constant function of $\lambda$.
    Then we can guarantee that $\norm{\hat{u} - u}^2 \leq O(\e)$ for $\e > 0$ small enough.
    Finally, by \cref{lem:clusteringalmosteigenvectors}, we can recover the partitions with at most $O(\e n)$ wrongly misclassified vertices up to a permutation.
\end{proof}
